Jenkinsfile- Basic CI Pipeline

pipeline {
    agent any
    stage {
	 stage('Build') {
	     steps {
		  echo ' Building project'
	      }
         }
         stage('Test') {
	     steps {
	    echo 'Running tests'
	      }
        }
        stage('Deploy') {
	    steps {
	    echo 'Deploying appication'
	    }
        }
    }
}

**************************************
Ansible : Install and configure Ngnix
**************************************

- name: Install and Configure Ngnix
  hosts: all
  become: yes
  tasks:
    - name: Install ngnix
      yum:
	name: ngnix
	state: present
    - name: Configure  ngnix
      template:
	src: nginx.conf.j2
	dest: /etc/ngnix/ngnix.conf
    - name: Start ngnix service
      service:
	name: ngnix
	state: started
	enabled: yes
