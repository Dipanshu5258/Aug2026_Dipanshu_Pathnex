Automating Cloud Infrastructure with Terraform & Ansible


**********************************************************
Ansible — Setup Nginx with Custom Config
**********************************************************
-name: Setupp Ngnix with Custmer config
 become: yes
 tasks:
   - name: Install ngnix
     yum:
	name: ngnix
	state: present
   -name: Copy custom ngnix.conf
    copy:
      scr: ngnix.conf
      dest: /etc/ngnix/ngnix.conf
   -name: Start ngnix service
    service:
      name: ngnix
      state: started 
      enabled: yes

**********************************************************
Terraform : Provision EC2 witth IAM Role
**********************************************************

resource "aws_iam_role" "ec2_role" {
  name		      = "pathnex_ec2_role"
  assume_role_policy  = jsonencode({
    Version ="2012-10-17"
    Statement = [
      {
	Action = "sts:AssumeRole"
	Effect = "Allow"
	Principal = {
	  Service = "ec2.amazonaws.com"
	}
      },
     ]
   })
}

resource "aws_instance" "pathnex_ec2" {
  ami			="ami-0abcd1234abcd1234"
  instance_type		="t2.medium"
  iam_instance_profile  =  aws_iam_role.ec2_role.name
  tags = {
    Name = "Pathnex-EC2"
  }
}
